# Changelog Generation

## Changelog Formats

```
Keep a Changelog  — Manual sections: Added, Changed, Fixed, etc.
Conventional      — Auto-generated from conventional commit types
GitHub Releases   — Auto-generated from PR titles + labels

Recommended: Conventional Changelog (auto from commit messages)
Format: Keep a Changelog structure (human-readable sections)
```

```toml
[changelog]
header = """
# Changelog\n
All notable changes to this project will be documented in this file.\n
"""
body = """
{% if version %}\
    ## [{{ version | trim_start_matches(pat="v") }}] - {{ timestamp | date(format="%Y-%m-%d") }}
{% else %}\
    ## [Unreleased]
{% endif %}\
{% for group, commits in commits | group_by(attribute="group") %}
    ### {{ group | striptags | trim | upper_first }}
    {% for commit in commits %}
        - **{{ commit.scope }}**: {{ commit.message | upper_first }}\
            {% if commit.breaking %} (**BREAKING**){% endif %}\
            {% if commit.pr %} (#{{ commit.pr }}){% endif %}
    {% endfor %}
{% endfor %}\n
"""
trim = true
footer = """
<!-- generated by git-cliff -->
"""

[git]
conventional_commits = true
filter_unconventional = true
split_commits = false
commit_parsers = [
    { message = "^feat", group = "<!-- 0 -->Features" },
    { message = "^fix", group = "<!-- 1 -->Bug Fixes" },
    { message = "^perf", group = "<!-- 2 -->Performance" },
    { message = "^doc", group = "<!-- 3 -->Documentation" },
    { message = "^refactor", group = "<!-- 4 -->Refactor" },
    { message = "^style", group = "<!-- 5 -->Style" },
    { message = "^test", group = "<!-- 6 -->Testing" },
    { message = "^chore\\(release\\)", skip = true },
    { message = "^chore|^ci", group = "<!-- 7 -->Miscellaneous" },
    { body = ".*security", group = "<!-- 8 -->Security" },
    { message = "BREAKING", group = "<!-- 9 -->BREAKING CHANGES" },
]
protect_breaking_commits = false
filter_commits = false
tag_pattern = "v[0-9].*"
sort_commits = "oldest"

[remote.github]
owner = "org"
repo = "quotation-app"
```

## Custom Changelog Generator

### TypeScript Generator

```typescript
// scripts/generate-changelog.ts
import { execFileSync } from 'child_process';

interface Commit {
    hash: string;
    type: string;
    scope: string;
    subject: string;
    body: string;
    breaking: boolean;
    pr: string | null;
}

interface GroupedCommits {
    features: Commit[];
    fixes: Commit[];
    performance: Commit[];
    breaking: Commit[];
    other: Commit[];
}

function getCommitsSinceTag(tag: string): Commit[] {
    const output = execFileSync('git', [
        'log', `${tag}..HEAD`, '--pretty=format:%H||%s||%b', '--no-merges',
    ], { encoding: 'utf-8' });

    return output.trim().split('\n').filter(Boolean).map(line => {
        const [hash, subject, body = ''] = line.split('||');
        const match = subject.match(/^(\w+)(?:\(([^)]+)\))?!?:\s*(.+)/);

        if (!match) {
            return {
                hash: hash.substring(0, 7),
                type: 'other',
                scope: '',
                subject: subject,
                body,
                breaking: false,
                pr: extractPr(body) || extractPr(subject),
            };
        }

        return {
            hash: hash.substring(0, 7),
            type: match[1],
            scope: match[2] || '',
            subject: match[3],
            body,
            breaking: subject.includes('!') || body.includes('BREAKING CHANGE'),
            pr: extractPr(body) || extractPr(subject),
        };
    });
}

function extractPr(text: string): string | null {
    const match = text.match(/#(\d+)/);
    return match ? match[1] : null;
}

function groupCommits(commits: Commit[]): GroupedCommits {
    const groups: GroupedCommits = {
        features: [],
        fixes: [],
        performance: [],
        breaking: [],
        other: [],
    };

    for (const commit of commits) {
        if (commit.breaking) {
            groups.breaking.push(commit);
        }
        switch (commit.type) {
            case 'feat': groups.features.push(commit); break;
            case 'fix': groups.fixes.push(commit); break;
            case 'perf': groups.performance.push(commit); break;
            default: groups.other.push(commit); break;
        }
    }

    return groups;
}

function formatCommit(commit: Commit): string {
    const scope = commit.scope ? `**${commit.scope}**: ` : '';
    const pr = commit.pr ? ` (#${commit.pr})` : '';
    return `- ${scope}${commit.subject}${pr}`;
}

function generateChangelog(version: string, previousTag: string): string {
    const commits = getCommitsSinceTag(previousTag);
    const groups = groupCommits(commits);
    const date = new Date().toISOString().split('T')[0];

    let changelog = `## [${version}] - ${date}\n\n`;

    if (groups.breaking.length > 0) {
        changelog += '### BREAKING CHANGES\n\n';
        for (const c of groups.breaking) {
            changelog += `${formatCommit(c)}\n`;
            if (c.body.includes('BREAKING CHANGE:')) {
                const desc = c.body.split('BREAKING CHANGE:')[1]?.trim();
                if (desc) changelog += `  > ${desc.split('\n')[0]}\n`;
            }
        }
        changelog += '\n';
    }

    if (groups.features.length > 0) {
        changelog += '### Features\n\n';
        for (const c of groups.features) changelog += `${formatCommit(c)}\n`;
        changelog += '\n';
    }

    if (groups.fixes.length > 0) {
        changelog += '### Bug Fixes\n\n';
        for (const c of groups.fixes) changelog += `${formatCommit(c)}\n`;
        changelog += '\n';
    }

    if (groups.performance.length > 0) {
        changelog += '### Performance\n\n';
        for (const c of groups.performance) changelog += `${formatCommit(c)}\n`;
        changelog += '\n';
    }

    return changelog;
}

// Run
const previousTag = process.argv[2] || execFileSync('git', [
    'describe', '--tags', '--abbrev=0',
], { encoding: 'utf-8' }).trim();

const version = process.argv[3] || JSON.parse(
    require('fs').readFileSync('package.json', 'utf-8')
).version;

const entry = generateChangelog(version, previousTag);
console.log(entry);
```

```json
        "changelog:generate": "npx tsx scripts/generate-changelog.ts"
    }
}
```

## GitHub Release Notes

### Auto-Generate from PR Labels

```yaml
# .github/workflows/release-notes.yml
name: Release Notes

on:
  push:
    tags: ['v*']

jobs:
  release-notes:
    runs-on: ubuntu-latest
    permissions:
      contents: write
      pull-requests: read
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Generate release notes
        id: notes
        run: |
          TAG=${GITHUB_REF#refs/tags/v}
          PREV_TAG=$(git describe --tags --abbrev=0 HEAD^ 2>/dev/null || echo "")

          {
            echo "body<<NOTES_EOF"

            if [ -n "$PREV_TAG" ]; then
              echo "## What's Changed"
              echo ""

              # Features
              FEATS=$(git log "${PREV_TAG}..HEAD" --pretty=format:"- %s (%h)" --grep="^feat" 2>/dev/null)
              if [ -n "$FEATS" ]; then
                echo "### Features"
                echo ""
                echo "$FEATS"
                echo ""
              fi

              # Bug fixes
              FIXES=$(git log "${PREV_TAG}..HEAD" --pretty=format:"- %s (%h)" --grep="^fix" 2>/dev/null)
              if [ -n "$FIXES" ]; then
                echo "### Bug Fixes"
                echo ""
                echo "$FIXES"
                echo ""
              fi

              # Performance
              PERFS=$(git log "${PREV_TAG}..HEAD" --pretty=format:"- %s (%h)" --grep="^perf" 2>/dev/null)
              if [ -n "$PERFS" ]; then
                echo "### Performance"
                echo ""
                echo "$PERFS"
                echo ""
              fi

              # Breaking changes
              BREAKS=$(git log "${PREV_TAG}..HEAD" --pretty=format:"- %s (%h)" --grep="BREAKING" 2>/dev/null)
              if [ -n "$BREAKS" ]; then
                echo "### BREAKING CHANGES"
                echo ""
                echo "$BREAKS"
                echo ""
              fi

              echo "---"
              echo ""
              echo "**Full Changelog**: https://github.com/${{ github.repository }}/compare/${PREV_TAG}...v${TAG}"
            else
              echo "Initial release"
            fi

            echo ""
            echo "### Installation"
            echo ""
            echo "\`\`\`bash"
            echo "npm install quotation-app@${TAG}"
            echo "\`\`\`"

            echo "NOTES_EOF"
          } >> $GITHUB_OUTPUT

      - name: Create GitHub Release
        uses: softprops/action-gh-release@v2
        with:
          body: ${{ steps.notes.outputs.body }}
          draft: false
          prerelease: ${{ contains(github.ref, '-alpha') || contains(github.ref, '-beta') || contains(github.ref, '-rc') }}
```

## Changelog Linting

### Enforce Conventional Commits

```bash
# Install
npm install -D @commitlint/cli @commitlint/config-conventional husky

# Configure
echo "export default { extends: ['@commitlint/config-conventional'] };" > commitlint.config.ts

# Setup hook
npx husky init
echo "npx --no -- commitlint --edit \$1" > .husky/commit-msg
```

### Validate Changelog in CI

```yaml
# .github/workflows/changelog-check.yml
name: Changelog Check

on:
  pull_request:
    branches: [main]

jobs:
  check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Check commit messages follow conventional format
        run: |
          npx commitlint --from origin/main --to HEAD --verbose

      - name: Verify changelog was updated (for non-trivial changes)
        run: |
          # Check if any source files changed
          CHANGED=$(git diff --name-only origin/main...HEAD | grep -c 'src/.*\.ts$' || true)

          if [ "$CHANGED" -gt 0 ]; then
            CHANGELOG_CHANGED=$(git diff --name-only origin/main...HEAD | grep -c 'CHANGELOG.md' || true)

            if [ "$CHANGELOG_CHANGED" -eq 0 ]; then
              echo "::warning::Source files changed but CHANGELOG.md was not updated"
            fi
          fi
```

### HTML Changelog Template

```typescript
// scripts/changelog-html.ts — generate HTML changelog for docs site
interface ChangelogEntry {
    version: string;
    date: string;
    sections: {
        title: string;
        items: string[];
    }[];
}

function generateHtmlChangelog(entries: ChangelogEntry[]): string {
    let html = `<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Changelog</title>
    <style>
        body { font-family: -apple-system, sans-serif; max-width: 800px; margin: 2rem auto; padding: 0 1rem; }
        h2 { border-bottom: 2px solid #e1e4e8; padding-bottom: 0.5rem; }
        h3 { color: #24292e; margin-top: 1.5rem; }
        ul { padding-left: 1.5rem; }
        li { margin: 0.3rem 0; }
        .breaking { color: #cb2431; font-weight: bold; }
        .version-tag { background: #0366d6; color: white; padding: 0.2rem 0.6rem; border-radius: 3px; font-size: 0.9rem; }
        .date { color: #586069; font-size: 0.9rem; }
    </style>
</head>
<body>
    <h1>Changelog</h1>
    <p>All notable changes to this project.</p>
`;

    for (const entry of entries) {
        html += `
    <h2>
        <span class="version-tag">${entry.version}</span>
        <span class="date">${entry.date}</span>
    </h2>
`;
        for (const section of entry.sections) {
            const isBreaking = section.title === 'BREAKING CHANGES';
            html += `    <h3${isBreaking ? ' class="breaking"' : ''}>${section.title}</h3>\n    <ul>\n`;
            for (const item of section.items) {
                html += `        <li>${item}</li>\n`;
            }
            html += `    </ul>\n`;
        }
    }

    html += `</body></html>`;
    return html;
}
```

## Code Style Rules

- CHANGELOG.md must be auto-generated — never manually write changelog entries
- Every commit that changes user-facing behavior must follow Conventional Commits format
- Use commitlint to enforce commit format — reject non-conforming commits in CI
- Changelog sections ordered: Breaking Changes → Features → Bug Fixes → Performance → Other
- Include issue/PR numbers in changelog entries for traceability (`#123`)
- Scope every commit (`feat(quotation):`, `fix(auth):`) so changelog entries are clear
- Generate release notes at the same time as CHANGELOG.md — not separately
- Keep `Unreleased` section at top during development, move to versioned on release
- For monorepos: generate per-package changelogs using changesets
- HTML changelog is for docs site only — CHANGELOG.md is the source of truth
